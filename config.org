  (defun my/org-noter-insert-categorized-note ()
    (interactive)

    (let* ((region
            (and (pdf-view-active-region-p)
                 (pdf-view-active-region)))

           (text
            (when region
              (mapconcat #'identity
                         (pdf-view-active-region-text)
                         " ")))

           (page (pdf-view-current-page))

           (pdf-buf (current-buffer))

           ;; janela PDF capturada antes do org-noter mudar o foco —
           ;; pdf-annot-add-markup-annotation exige a janela selecionada
           (pdf-win (selected-window)))

      ;; Fase 2: cria nota com região ainda ativa
      (let ((org-noter-insert-note-no-questions t))
        (org-noter-insert-precise-note))

      ;; Fase 3: prompts
      (let* ((category
              (completing-read "Category: "
                               my/org-noter-categories
                               nil t))

             (title
              (read-string
               (format "Title (default: %s): " category)))

             (final-title
              (if (string-empty-p title) category title))

             (color
              (alist-get category
                         my/org-noter-category-colors
                         nil nil #'string=)))

        ;; Fase 4: cria highlight com a cor da categoria.
        ;; with-selected-window é necessário: pdf-annot-add-markup-annotation
        ;; verifica a janela selecionada (não o current-buffer).
        (when (and region color (window-live-p pdf-win))
          (with-selected-window pdf-win
            (pdf-annot-add-markup-annotation region 'highlight color)))

        ;; Fase 5: edita heading, abre tags, insere conteúdo
        ;; with-selected-window garante buffer e ponto da janela corretos
        (let ((notes-win (org-noter--get-notes-window)))
          (when notes-win
            (with-selected-window notes-win
              (org-back-to-heading t)
              (org-edit-headline final-title)
              (condition-case nil (org-set-tags-command) (quit nil))
              (org-end-of-meta-data t)
              (when (and text (not (string-empty-p text)))
                (insert "\n#+begin_quote\n"
                        (string-trim-right text)
                        (if page (format " (p. %d)" page) "")
                        "\n#+end_quote\n"))
              (insert "\n")))))))

   :map org-noter-doc-mode-map
   :localleader
   :desc "Categorized note" "a" #'my/org-noter-insert-categorized-note
   :desc "Kill session"     "Q" #'org-noter-kill-session)
